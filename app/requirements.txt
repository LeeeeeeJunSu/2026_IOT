# Runtime dependencies shared by Windows/Linux/Raspberry Pi.
#
# Install from the repository root:
#   python3 -m venv .venv-pi
#   source .venv-pi/bin/activate
#   python -m pip install --upgrade pip
#   python -m pip install -r requirements.txt
#   python -m pip install -r requirements-pi-cpu.txt
#
# Recommended system packages for the full app:
#   sudo apt install python3-pip python3-venv python3-tk python3-gpiozero python3-lgpio
#
# Tkinter is required for the desktop GUI and comes from `python3-tk`.
# Raspberry Pi 5 GPIO LED control uses gpiozero with the lgpio backend.
# Installing python3-gpiozero and python3-lgpio from apt is recommended.
# The web dashboard and raw-data fallback can run without Tkinter.
# Keep scikit-learn pinned because saved model_bundle.pkl files are
# version-sensitive.
# PyTorch is kept in a separate requirements file so Raspberry Pi can install
# CPU-only wheels and CUDA workstations can use the matching CUDA wheel.

numpy==2.4.4
scipy==1.17.1
scikit-learn==1.8.0
joblib==1.5.3
threadpoolctl==3.6.0
